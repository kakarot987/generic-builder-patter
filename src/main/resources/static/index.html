// Pizza.java — the Product (immutable)
public class Pizza {

  private final String size, crust, sauce;
  private final String cheese, toppings;
  private final boolean extraCheese, wellDone;

  // Only Builder can call this
  private Pizza(Builder b) {
    this.size        = b.size;
    this.crust       = b.crust;
    this.sauce       = b.sauce;
    this.cheese      = b.cheese;
    this.toppings    = b.toppings;
    this.extraCheese = b.extraCheese;
    this.wellDone    = b.wellDone;
  }

  // Getters only — no setters
  public String  getSize()        { return size; }
  public String  getCrust()       { return crust; }
  public boolean isExtraCheese() { return extraCheese; }
  // ... etc
}
// Static inner Builder class
public static class Builder {

  // Required — passed in constructor
  private final String size;
  private final String cheese;

  // Optional — have sensible defaults
  private String  crust       = "Classic";
  private String  sauce       = "Tomato";
  private String  toppings    = "";
  private boolean extraCheese = false;
  private boolean wellDone    = false;

  public Builder(String size, String cheese) {
    this.size   = size;
    this.cheese = cheese;
  }

  /* Each setter returns `this` → fluent chain */
  public Builder crust(String c)    { crust = c; return this; }
  public Builder sauce(String s)    { sauce = s; return this; }
  public Builder toppings(String t) { toppings = t; return this; }
  public Builder extraCheese()     { extraCheese = true; return this; }
  public Builder wellDone()        { wellDone = true; return this; }

  // Terminal — validates then constructs
  public Pizza build() {
    return new Pizza(this);
  }
}
@RestController
@RequestMapping("/api/pizza")
@CrossOrigin(origins = "*")
public class PizzaController {

  @PostMapping("/build")
  public ResponseEntity<Pizza> build(
          @RequestBody PizzaRequest req) {

    Pizza.Builder b = new Pizza.Builder(
            req.getSize(), req.getCheese())
        .crust(req.getCrust())
        .sauce(req.getSauce())
        .toppings(req.getToppings());

    if (req.isExtraCheese()) b.extraCheese();
    if (req.isWellDone())    b.wellDone();

    return ResponseEntity.ok(b.build());
  }
}

Why Builder?
Avoids telescoping constructors with 7+ positional params. Fluent API is self-documenting. Required vs optional fields are explicit. Validates before constructing. Result is immutable — thread-safe by default.
